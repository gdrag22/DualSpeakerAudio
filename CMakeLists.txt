cmake_minimum_required(VERSION 3.20)
project(DualMonitorAudio VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "DualMonitorAudio requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(DualMonitorAudio src/main.cpp)
target_compile_definitions(DualMonitorAudio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(DualMonitorAudio PRIVATE ole32 uuid avrt shell32)
if(MSVC)
  target_compile_options(DualMonitorAudio PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(DualMonitorAudio PRIVATE /MANIFEST:EMBED /MANIFESTUAC:level='asInvoker')
else()
  target_compile_options(DualMonitorAudio PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(DualMonitorAudio PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
option(DUAL_AUDIO_TESTS "Build buffer and drift tests" OFF)
if(DUAL_AUDIO_TESTS)
  enable_testing()
  add_executable(AudioBufferTests tests/audio_buffer_tests.cpp)
  add_test(NAME AudioBufferTests COMMAND AudioBufferTests)
  add_executable(FormatTests tests/format_tests.cpp)
  target_compile_definitions(FormatTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(FormatTests PRIVATE uuid)
  add_test(NAME FormatTests COMMAND FormatTests)
  add_executable(NotificationTests tests/notification_tests.cpp)
  target_compile_definitions(NotificationTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(NotificationTests PRIVATE uuid)
  add_test(NAME NotificationTests COMMAND NotificationTests)
endif()
install(TARGETS DualMonitorAudio RUNTIME DESTINATION .)
